    % Bottom-left tile (color index 1 of region)
    % -----------------------------------------------------

    % Bottom-right tile (color index 2)
    % -----------------------------------------------------

    % -----------------------------------------------------
    % Top-left tile (color index 3)
    % -----------------------------------------------------
    \fill[\baGetColor{3}] (0,1) rectangle (1,2);

    % -----------------------------------------------------
    % Top-right tile (color index 4)
    % -----------------------------------------------------
    \fill[\baGetColor{4}] (1,1) rectangle (2,2);

    \draw (1,0) -- (1,2);
