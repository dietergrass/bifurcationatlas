% =========================================================
% PIC TEMPLATE DEFINITION
% =========================================================
% This defines a reusable TikZ picture called "Test".
%
% Key idea:
% - This is a template, not a fixed drawing
% - Colors are injected dynamically via region selection
% - \baGetColor{1..4} selects region-dependent colors
%
% Each region provides a different color configuration.
% =========================================================
\baNewPic{Test}{
	
    % -----------------------------------------------------
    % Bottom-left tile (color index 1 of region)
    % -----------------------------------------------------
    \fill[\baGetColor{1}] (0,0) rectangle (1,1);

    % -----------------------------------------------------
    % Bottom-right tile (color index 2)
    % -----------------------------------------------------
    \fill[\baGetColor{2}] (1,0) rectangle (2,1);

    % -----------------------------------------------------
    % Top-left tile (color index 3)
    % -----------------------------------------------------
    \fill[\baGetColor{3}] (0,1) rectangle (1,2);

    % -----------------------------------------------------
    % Top-right tile (color index 4)
    % -----------------------------------------------------
    \fill[\baGetColor{4}] (1,1) rectangle (2,2);

    % -----------------------------------------------------
    % Structural grid overlay
    % -----------------------------------------------------
    \draw[line width=2pt,black] (0,2) -- (0,0) -- (2,0);
    \draw (0,1) -- (2,1);
    \draw (1,0) -- (1,2);

    % -----------------------------------------------------
    % Optional reference axis highlight
    % -----------------------------------------------------
    \draw[line width=2pt,red] (0,0.04) -- (0,2);
}
\baNewPic{Cartel}{
{


    % ====================================================
    % 4 LARGE SQUARES (base layer)
    % each is 1x1
    % ====================================================

    % bottom-left

    % bottom-right
    \fill[\baGetColor{2}] (1,0) rectangle (2,1);

    % top-left
    \fill[\baGetColor{3}] (0,1) rectangle (1,2);

    % top-right
    \fill[\baGetColor{4}] (1,1) rectangle (2,2);

    % ====================================================
    % SMALL SQUARE (overwrites bottom-left)
    % drawn LAST = highest priority
    % ====================================================
    \fill[\baGetColor{5}] (0,0) rectangle (0.5,0.5);
    
    \draw[line width=2pt,black](0,2) -- (0,0) -- (2,0);
    \draw (0,1)--(2,1);
    \draw (1,0)--(1,2);
    \draw (0,0.5)--(0.5,0.5);
    \draw (0.5,0)--(0.5,0.5);
    \draw[line width=2pt,red] (0,0.04)--(0,2);
    
}
}
